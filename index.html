function doPost(e) { const data = JSON.parse(e.postData.contents); const tabName = data.entryBy === 'Dan' ? 'Dan' : 'Hussein'; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(tabName); if (!sheet) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: 'Missing sheet tab: ' + tabName })) .setMimeType(ContentService.MimeType.JSON); } sheet.appendRow([ data.entryBy || '', data.tastingDate || '', data.repImporter || '', data.producer || '', data.wineName || '', data.category || '', data.variety || '', data.acid || '', data.body || '', data.tannins || '', data.sweetness || '', data.condition || '', data.flawType || '', data.rating || '', data.placement || '', Array.isArray(data.techniques) ? data.techniques.join(', ') : '', data.price || '', data.quickTag || '', data.notes || '', data.createdAt || '' ]); return ContentService .createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); }
